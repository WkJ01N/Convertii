cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(Convertii VERSION 0.2.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CONVERTII_PUBLISHER "CN=Convertii" CACHE STRING "Identity package publisher")
set(CONVERTII_RELEASE_CONFIG "${CMAKE_SOURCE_DIR}/packaging/release.dev.json" CACHE FILEPATH "Installer release configuration")
set(CONVERTII_PAYLOAD "" CACHE FILEPATH "Optional offline installer ZIP")
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Convertii targets Windows x64 using MSVC and Windows SDK 10.0.26100.")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
add_library(miniz STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tdef.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(miniz PUBLIC third_party/miniz)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(ConvertiiCore STATIC native/core/platform.cpp native/core/media.cpp native/core/processor.cpp native/core/subtitles.cpp native/core/settings.cpp native/core/install.cpp native/core/integration.cpp)
target_include_directories(ConvertiiCore PUBLIC native/include third_party)
target_link_libraries(ConvertiiCore PUBLIC miniz bcrypt advapi32 shell32 shlwapi ole32 winhttp windowsapp version)
configure_file(native/app/app.manifest.in "${CMAKE_BINARY_DIR}/app.manifest" @ONLY)
set(CONVERTII_ICON "${CMAKE_SOURCE_DIR}/native/app/Convertii.ico")
file(TO_CMAKE_PATH "${CONVERTII_ICON}" CONVERTII_ICON_RC)
configure_file(native/app/app.rc.in "${CMAKE_BINARY_DIR}/app.rc" @ONLY)
add_executable(Convertii WIN32 native/app/main.cpp native/app/requests.cpp native/app/window.cpp native/app/player.cpp "${CMAKE_BINARY_DIR}/app.rc")
target_include_directories(Convertii PRIVATE native/app)
target_link_libraries(Convertii PRIVATE ConvertiiCore comctl32 comdlg32 gdi32 user32 uxtheme mfplat mf mfuuid mfreadwrite d3d11 dcomp dxgi)
target_link_options(Convertii PRIVATE /MANIFEST:NO)
file(TO_CMAKE_PATH "${CONVERTII_RELEASE_CONFIG}" CONVERTII_RELEASE_CONFIG_RC)
set(PAYLOAD_RESOURCE "")
if(CONVERTII_PAYLOAD)
 file(TO_CMAKE_PATH "${CONVERTII_PAYLOAD}" CONVERTII_PAYLOAD_RC)
 set(PAYLOAD_RESOURCE "102 RCDATA \"${CONVERTII_PAYLOAD_RC}\"")
endif()
configure_file(native/setup/setup.rc.in "${CMAKE_BINARY_DIR}/setup.rc" @ONLY)
# Repacking the icon must relink both executables, not only be picked up later.
set_source_files_properties("${CMAKE_BINARY_DIR}/app.rc" PROPERTIES OBJECT_DEPENDS "${CONVERTII_ICON}")
set_source_files_properties("${CMAKE_BINARY_DIR}/setup.rc" PROPERTIES OBJECT_DEPENDS "${CONVERTII_ICON}")
add_executable(ConvertiiSetup WIN32 native/setup/main.cpp "${CMAKE_BINARY_DIR}/setup.rc")
target_link_libraries(ConvertiiSetup PRIVATE ConvertiiCore comctl32 comdlg32 user32 gdi32)
target_link_options(ConvertiiSetup PRIVATE /MANIFEST:NO)
add_subdirectory(native/Convertii.Shell)
set_target_properties(ConvertiiShell ShellSmokeTest ShellRequestSink PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/shell/$<CONFIG>")
include(CTest)
if(BUILD_TESTING)
 add_executable(CoreTests native/tests/core_tests.cpp)
 target_link_libraries(CoreTests PRIVATE ConvertiiCore)
 add_executable(HardwareStub native/tests/hardware_stub.cpp)
 target_link_libraries(HardwareStub PRIVATE ConvertiiCore)
 add_dependencies(CoreTests HardwareStub)
 add_executable(InstallTests native/tests/install_tests.cpp)
 target_link_libraries(InstallTests PRIVATE ConvertiiCore)
 add_executable(IconTests native/tests/icon_tests.cpp)
 configure_file(native/tests/ui.rc.in "${CMAKE_BINARY_DIR}/ui.rc" @ONLY)
 add_executable(UiTests native/tests/ui_tests.cpp native/app/requests.cpp native/app/window.cpp native/app/player.cpp "${CMAKE_BINARY_DIR}/ui.rc")
 target_link_options(UiTests PRIVATE /MANIFEST:NO)
 target_include_directories(UiTests PRIVATE native/app)
 target_link_libraries(UiTests PRIVATE ConvertiiCore comctl32 comdlg32 gdi32 user32 uxtheme mfplat mf mfuuid mfreadwrite d3d11 dcomp dxgi windowscodecs psapi)
 add_test(NAME ui COMMAND UiTests)
 add_test(NAME core COMMAND CoreTests)
 add_test(NAME install COMMAND InstallTests)
 add_test(NAME shell COMMAND ShellSmokeTest "$<TARGET_FILE:ConvertiiShell>")
 add_test(NAME icon COMMAND IconTests "$<TARGET_FILE:Convertii>")
 set_tests_properties(core install shell icon ui PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 240)
endif()

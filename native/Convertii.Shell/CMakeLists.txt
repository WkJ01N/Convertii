cmake_minimum_required(VERSION 3.21)
project(ConvertiiShell LANGUAGES CXX)
add_library(ConvertiiShell SHARED Shell.cpp Shell.def)
target_compile_features(ConvertiiShell PRIVATE cxx_std_17)
target_compile_definitions(ConvertiiShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ConvertiiShell PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_include_directories(ConvertiiShell PRIVATE "${CMAKE_SOURCE_DIR}/third_party" "${CMAKE_SOURCE_DIR}/native/include")
target_link_libraries(ConvertiiShell PRIVATE shlwapi shell32 ole32 advapi32)
set_property(TARGET ConvertiiShell PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ShellSmokeTest SmokeTest.cpp)
target_compile_features(ShellSmokeTest PRIVATE cxx_std_17)
target_compile_definitions(ShellSmokeTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(ShellSmokeTest PRIVATE shell32 ole32 user32)
target_compile_options(ShellSmokeTest PRIVATE /utf-8 /W4)
set_property(TARGET ShellSmokeTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ShellRequestSink WIN32 RequestSink.cpp)
set_target_properties(ShellRequestSink PROPERTIES OUTPUT_NAME "Convertii")
target_compile_definitions(ShellRequestSink PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(ShellRequestSink PRIVATE shell32)
target_compile_options(ShellRequestSink PRIVATE /utf-8 /W4)
set_property(TARGET ShellRequestSink PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
